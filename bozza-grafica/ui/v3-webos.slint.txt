component LiveScreen inherits Rectangle {
    in property <[Channel]> channels;
    in property <[string]> categories;
    in-out property <int> selected;
    in-out property <int> category: 0;
    callback play();
    callback pip();
    property <Channel> cur: root.channels[root.selected];

    function ensure-visible() {
        if (root.selected * 52px + fl.viewport-y < 0px) {
            fl.viewport-y = -root.selected * 52px;
        } else if ((root.selected + 1) * 52px + fl.viewport-y > fl.height) {
            fl.viewport-y = fl.height - (root.selected + 1) * 52px;
        }
    }
    init => { fs.focus(); }

    fs := FocusScope {
        key-pressed(event) => {
            if (event.text == Key.UpArrow) {
                root.selected = max(0, root.selected - 1);
                root.ensure-visible();
                return accept;
            }
            if (event.text == Key.DownArrow) {
                root.selected = min(root.channels.length - 1, root.selected + 1);
                root.ensure-visible();
                return accept;
            }
            if (event.text == Key.Return) {
                root.play();
                return accept;
            }
            reject
        }
        HorizontalLayout {
            padding-left: 44px;
            padding-right: 44px;
            padding-top: 8px;
            spacing: 44px;
            VerticalLayout {
                width: 520px;
                spacing: 10px;
                Tabs {
                    options: root.categories;
                    current <=> root.category;
                }
                Rectangle {
                    vertical-stretch: 1;
                    clip: true;
                    fl := Flickable {
                        width: parent.width;
                        height: parent.height;
                        viewport-height: root.channels.length * 52px;
                        for c[i] in root.channels: ChannelLine {
                            y: i * 52px;
                            width: fl.width;
                            c: c;
                            active: i == root.selected;
                            clicked => {
                                root.selected = i;
                                fs.focus();
                            }
                            activated => {
                                root.selected = i;
                                root.play();
                            }
                        }
                    }
                }
            }
            VerticalLayout {
                horizontal-stretch: 1;
                alignment: start;
                spacing: 6px;
                Preview {
                    height: 250px;
                    c: root.cur;
                }
                Rectangle { height: 10px; }
                Text {
                    text: root.cur.name;
                    color: Theme.ink;
                    font-size: 32px;
                    font-weight: 700;
                }
                Text {
                    text: root.cur.now;
                    color: Theme.ink;
                    font-size: 18px;
                }
                Text {
                    text: root.cur.start + " – " + root.cur.end + (root.cur.hd ? "  ·  HD" : "");
                    color: Theme.dim;
                    font-size: 13px;
                }
                Progress { value: root.cur.progress; }
                Text {
                    text: root.cur.desc;
                    color: Theme.dim;
                    font-size: 14px;
                    wrap: word-wrap;
                }
                Text {
                    text: "Dopo, alle " + root.cur.next-time + ": " + root.cur.next;
                    color: Theme.faint;
                    font-size: 13px;
                }
                Rectangle { height: 12px; }
                HorizontalLayout {
                    spacing: 12px;
                    alignment: start;
                    Btn { label: "Guarda"; primary: true; clicked => { root.play(); } }
                    Btn { label: "Mini player"; clicked => { root.pip(); } }
                    Btn { label: root.cur.fav ? "Tra i preferiti" : "Aggiungi ai preferiti"; }
                }
            }
        }
    }
}

component PlayerScreen inherits Rectangle {
    in property <[Channel]> channels;
    in property <[string]> categories;
    in-out property <int> selected;
    callback back();
    callback pip();
    property <Channel> c: root.channels[root.selected];
    property <bool> overlay: true;
    property <bool> drawer: false;
    property <int> cursor: 0;
    property <int> category: 0;
    background: #000000;

    function ensure-visible() {
        if (root.cursor * 52px + dl.viewport-y < 0px) {
            dl.viewport-y = -root.cursor * 52px;
        } else if ((root.cursor + 1) * 52px + dl.viewport-y > dl.height) {
            dl.viewport-y = dl.height - (root.cursor + 1) * 52px;
        }
    }
    function toggle-drawer() {
        root.drawer = !root.drawer;
        if (root.drawer) {
            root.cursor = root.selected;
            root.ensure-visible();
        }
    }
    init => { fs.focus(); }

    fs := FocusScope {
        key-pressed(event) => {
            if (event.text == Key.UpArrow) {
                if (root.drawer) {
                    root.cursor = max(0, root.cursor - 1);
                    root.ensure-visible();
                } else {
                    root.selected = max(0, root.selected - 1);
                    root.overlay = true;
                }
                return accept;
            }
            if (event.text == Key.DownArrow) {
                if (root.drawer) {
                    root.cursor = min(root.channels.length - 1, root.cursor + 1);
                    root.ensure-visible();
                } else {
                    root.selected = min(root.channels.length - 1, root.selected + 1);
                    root.overlay = true;
                }
                return accept;
            }
            if (event.text == Key.Return) {
                if (root.drawer) { root.selected = root.cursor; } else { root.toggle-drawer(); }
                return accept;
            }
            if (event.text == Key.Escape) {
                if (root.drawer) { root.drawer = false; } else { root.back(); }
                return accept;
            }
            if (event.text == "p") {
                root.pip();
                return accept;
            }
            if (event.text == " ") {
                root.overlay = !root.overlay;
                return accept;
            }
            reject
        }
    }

    // il video (qui libmpv disegnerà sotto l'interfaccia)
    Rectangle {
        background: @linear-gradient(160deg, root.c.tint.darker(0.45) 0%, #05060f 80%);
    }
    Text {
        text: root.c.num;
        color: #ffffff14;
        font-size: 300px;
        font-weight: 200;
    }
    TouchArea {
        pointer-event(e) => {
            if (e.button == PointerEventButton.right && e.kind == PointerEventKind.up) {
                root.toggle-drawer();
                fs.focus();
            }
        }
        clicked => {
            if (root.drawer) { root.drawer = false; } else { root.overlay = !root.overlay; }
            fs.focus();
        }
        double-clicked => { root.back(); }
    }

    // barra in basso
    Rectangle {
        y: parent.height - self.height;
        height: 190px;
        opacity: root.overlay ? 1 : 0;
        animate opacity { duration: 200ms; }
        background: @linear-gradient(180deg, #00000000 0%, #000000d0 100%);
        HorizontalLayout {
            padding: 40px;

                HorizontalLayout {

                        color: Theme.dim;

                    }

                    Btn { label: "Audio"; }

                }
            }
        }
    }
    Text {
        x: 40px;
        y: 30px;
        text: "Clic destro sul video: elenco canali  ·  P: mini player";
        color: #ffffff90;
        font-size: 14px;
        opacity: root.overlay && !root.drawer ? 1 : 0;
        animate opacity { duration: 200ms; }
    }

    // l'elenco canali che scorre da sinistra
    Rectangle {
        x: root.drawer ? 0px : -self.width - 20px;
        width: 490px;
